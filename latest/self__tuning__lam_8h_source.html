150 [[nodiscard]] Solution Run(std::size_t evaluations) {
151 // Self-tuning
152 const std::size_t n = std::max(evaluations, kMinEvaluations);
153 const std::size_t m = TuningLength(n);
154 const double alpha = TuneAlpha(n);
155 double accept_rate = InitialAcceptanceRate(n);
156 auto [s, temperature, beta] = TuneTemperature(n, m, accept_rate);
157
158 double m0 = accept_rate - 0.44;
159 const double m1 = std::pow(560.0, -1.0 / (0.15 * static_cast<double>(n)));
160 const double m2 = std::pow(440.0, -1.0 / (0.35 * static_cast<double>(n)));
161 double lam_rate = accept_rate;
162
163 const std::size_t early_end =
164 static_cast<std::size_t>(0.15 * static_cast<double>(n));
165 const std::size_t late_start =
166 static_cast<std::size_t>(0.65 * static_cast<double>(n));
167
168 for (std::size_t i = m + 1; i <= n; ++i) {
169 const double c0 = s.cost;
170 auto p = problem_.Propose(s, rng_);
171 const double& c1 = s.cost;
172 if (c1 <= c0 || uniform_(rng_) < std::exp(-(c1 - c0) / temperature)) {
173 problem_.Accept(s, std::move(p));
174 accept_rate = (1.0 - alpha) * accept_rate + alpha;
175 Consider(s);
176 } else {
177 problem_.Reject(s, std::move(p));
178 accept_rate *= 1.0 - alpha;
179 }
180
181 if (i <= early_end) {
182 m0 = m0 * m1;
183 lam_rate = 0.44 + m0;
184 } else if (i > late_start) {
185 lam_rate *= m2;
186 } else {
187 lam_rate = 0.44;
188 }
189
190 if (accept_rate > lam_rate) {
191 temperature *= beta;
192 } else {
193 temperature /= beta;
194 }
195 }
196
197 return std::move(best_);
198 }

203 Solution solution;
204 double t0;
205 double beta;

210 static std::size_t TuningLength(std::size_t n) {
211 return (n >= 10000u) ? n / 1000 : n / 100;
212 }

216 static double TuneAlpha(std::size_t n) {
217 return std::min(2.0 / (0.01 * static_cast<double>(n) + 1.0), 0.2);
218 }

222 static double InitialAcceptanceRate(std::size_t n) {
223 return (n >= 10000u) ? kLambdaAt0p001 : kLambdaAt0p01;

237 Tuned TuneTemperature(std::size_t n, std::size_t m, double initial_rate) {
238 const bool long_run = n >= 10000u;
239 const double& zeta1 = long_run ? kZetaAt0p001 : kZetaAt0p01;
240 const double& zeta2 = long_run ? kZetaAt0p002 : kZetaAt0p02;
241 const double& lambda2 = long_run ? kLambdaAt0p002 : kLambdaAt0p02;
242
243 double sum = 0.0; // SumOfDifferences in paper
244 std::size_t diff = 0; // k in paper
245 std::size_t non_worse = 0; // d in paper
246 Solution s = problem_.RandomSolution(rng_);
247 best_ = s; // Initialize best_ to the initial solution.
248 for (std::size_t i = 1; i <= m; ++i) {
249 const double c0 = s.cost;
250 auto p = problem_.Propose(s, rng_);
251 const double c1 = s.cost;
252 if (c1 != c0) {
253 sum += std::abs(c1 - c0);
254 diff += 1;
255 }
256 if (c1 <= c0) {
257 non_worse += 1;
258 }
259 problem_.TuneAccept(s, std::move(p));
260 Consider(s);
261 }
262
263 const double delta_c = diff != 0 ? sum / static_cast<double>(diff) : 1.0;
264 const double gamma = static_cast<double>(non_worse) /
265 static_cast<double>(m + (non_worse == m));
266 const double& r = lambda2;
267 double t0, beta;
268
269 if (gamma < initial_rate) {
270 const double l0 = std::log((initial_rate - gamma) / (1.0 - gamma));
271 t0 = -delta_c / l0;
272 if (gamma < r) {
273 const double lr = std::log((r - gamma) / (1.0 - gamma));
274 beta = std::pow(l0 / lr, 1.0 / static_cast<double>(m));
275 } else {
276 beta = std::pow(-zeta2 * l0, 1.0 / static_cast<double>(m));
277 }
278 } else {
279 t0 = delta_c * zeta1;
280 beta = std::pow(zeta2 / zeta1, 1.0 / static_cast<double>(m));
281 }
282
283 return {std::move(s), t0, beta};
284 }
285
287 void Consider(const Solution& s) {
288 if (s.cost < best_.cost) best_ = s;
289 }
